# list-operators
list operators 
numbers = [10, 20, 30, 40, 50]

print("Original List:", numbers)

# Find sum
print("Sum:", sum(numbers))

# Find maximum
print("Maximum:", max(numbers))

# Find minimum
print("Minimum:", min(numbers))

# Find length
print("Length:", len(numbers))

# Reverse the list
print("Reversed List:", numbers[::-1])
